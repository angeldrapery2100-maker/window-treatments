# Allow the public site-settings XHR so Googlebot can render SiteNav
# correctly during indexing — SiteNav fetches this on hydration to decide
# whether the "Online Store" link is shown. The longer-path Allow wins
# over the shorter-path Disallow: /api below. The response itself sets
# X-Robots-Tag: noindex so the JSON isn't indexed as its own document.
Allow: /api/site-settings

# Same reasoning for the fabric library: /fabrics renders its grid from
# /api/fabrics on hydration, so blocking that endpoint leaves Googlebot
# looking at an empty page. The individual /fabrics/<id> pages are
# server-rendered and need no exception.
Allow: /api/fabrics

Disallow: /admin
Disallow: /api
Disallow: /store/cart
Disallow: /store/checkout
Disallow: /store/account
Disallow: /store/order-confirmation
