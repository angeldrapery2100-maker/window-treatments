    <script>

        const slider = document.getElementById('slider');

        const blueFill = document.getElementById('blue-fill');

        const percentText = document.getElementById('percent-text');

        const curtainFabric = document.getElementById('curtain-fabric');

        const bottomBar = document.getElementById('bottom-bar');


        function updateShade(openPercent) {

            // openPercent: 100 是全开, 0 是全关

            const shadeDroppedPercent = 100 - openPercent; // 遮挡比例

            

            // 更新滑块

            blueFill.style.height = shadeDroppedPercent + '%';

            percentText.innerText = openPercent;


            // 更新卷帘 SVG (总高度 375)

            const curtainHeight = (375 * shadeDroppedPercent) / 100;

            curtainFabric.setAttribute('height', curtainHeight);

            bottomBar.setAttribute('y', 90 + curtainHeight);

        }


        // 滑动点击交互

        slider.addEventListener('mousedown', function(e) {

            const handleMove = (moveEvent) => {

                const rect = slider.getBoundingClientRect();

                const y = moveEvent.clientY - rect.top;

                const percent = 100 - Math.max(0, Math.min(100, Math.round((y / rect.height) * 100)));

                updateShade(percent);

            };

            

            handleMove(e);

            window.addEventListener('mousemove', handleMove);

            window.addEventListener('mouseup', () => window.removeEventListener('mousemove', handleMove), { once: true });

        });

    </script>
